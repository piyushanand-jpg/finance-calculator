    <h2>What is an EMI?</h2>
    <p>EMI stands for Equated Monthly Installment — the fixed amount you pay your lender every month until your loan is fully repaid. Each EMI includes two parts: a portion that reduces your principal (the amount you borrowed) and a portion that pays interest to the lender. Early in the loan, most of your EMI goes toward interest; toward the end, most goes toward principal.</p>

    <h2>How is EMI calculated?</h2>
    <p>Lenders use a standard formula based on your loan amount, interest rate, and tenure:</p>
    <p style="font-family:'Fraunces',serif; font-size:17px; color:var(--navy); background:var(--gold-soft); padding:14px 18px; border-radius:10px; display:inline-block;">EMI = P × r × (1+r)^n / ((1+r)^n − 1)</p>
    <p>Where P is the loan amount, r is the monthly interest rate, and n is the number of monthly installments. You don't need to do this by hand — the calculator above does it instantly as you move the sliders.</p>

    <h2>Tips to reduce your EMI burden</h2>
    <ul>
      <li>Make a larger down payment to reduce the loan amount, and therefore the EMI.</li>
      <li>Choose a longer tenure to lower your monthly EMI — but remember this increases total interest paid.</li>
      <li>Compare interest rates across lenders before finalizing your loan.</li>
      <li>Consider prepaying part of the loan when you have surplus funds, which reduces total interest.</li>
    </ul>

<script>
  const amountEl = document.getElementById('amount');
  const rateEl = document.getElementById('rate');
  const tenureEl = document.getElementById('tenure');

  const amountVal = document.getElementById('amountVal');
  const rateVal = document.getElementById('rateVal');
  const tenureVal = document.getElementById('tenureVal');

  const emiOut = document.getElementById('emiOut');
  const totalInterestOut = document.getElementById('totalInterestOut');
  const totalPaymentOut = document.getElementById('totalPaymentOut');
  const splitPrincipal = document.getElementById('splitPrincipal');
  const splitInterest = document.getElementById('splitInterest');
  const splitPct = document.getElementById('splitPct');

  function formatINR(num){
    return '₹' + Math.round(num).toLocaleString('en-IN');
  }

  function calculate(){
    const P = parseFloat(amountEl.value);
    const annualRate = parseFloat(rateEl.value);
    const years = parseInt(tenureEl.value);

    const r = annualRate / 12 / 100;
    const n = years * 12;

    let emi;
    if (r === 0) {
      emi = P / n;
    } else {
      emi = P * r * Math.pow(1 + r, n) / (Math.pow(1 + r, n) - 1);
    }

    const totalPayment = emi * n;
    const totalInterest = totalPayment - P;

    amountVal.textContent = formatINR(P);
    rateVal.textContent = annualRate.toFixed(1) + '%';
    tenureVal.textContent = years + ' yrs';

    emiOut.textContent = formatINR(emi);
    totalInterestOut.textContent = formatINR(totalInterest);
    totalPaymentOut.textContent = formatINR(totalPayment);

    const principalPct = (P / totalPayment) * 100;
    const interestPct = 100 - principalPct;

    splitPrincipal.style.width = principalPct + '%';
    splitInterest.style.width = interestPct + '%';
    splitPct.textContent = principalPct.toFixed(0) + '% / ' + interestPct.toFixed(0) + '%';
  }

  [amountEl, rateEl, tenureEl].forEach(el => el.addEventListener('input', calculate));
  calculate();
</script>
